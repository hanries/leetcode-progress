class Solution:
   
    def isBalanced(self, root: TreeNode | None) -> bool:
        if root is None:
            return True
        depthL = self.helper(root.left)
        depthR = self.helper(root.right)
        if abs(depthL-depthR) > 1:
            return False
        return self.isBalanced(root.left) and self.isBalanced(root.right)
    
    def helper(self, root: TreeNode) -> int:
        if root is None:
            return 0
        depthL = self.helper(root.left) + 1
        depthR = self.helper(root.right) + 1
        return max(depthL, depthR)
