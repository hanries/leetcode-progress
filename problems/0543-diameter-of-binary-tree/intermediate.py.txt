    def maxDiameter(self,root):
        if root is None:
            return 0
        leftDepth = self.maxDiameter(root.left)[0]
        rightDepth = self.maxDiameter(root.right)[0]
        leftDiameter = self.maxDiameter(root.left)[1]
        rightDiameter = self.maxDiameter(root.right)[1]
        diameter = leftDepth + rightDepth

        return max(leftDepth,rightDepth) + 1 , max(diameter,leftDiameter,rightDiameter)
